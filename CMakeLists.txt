cmake_minimum_required(VERSION 3.22)
project(iu9-neural-networks LANGUAGES CXX)
include(CTest)
option(NN_ENABLE_PLOTS "Enable interactive Matplot++ plots" OFF)
option(NN_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(spdlog REQUIRED)
if(NN_ENABLE_PLOTS)
  find_package(Matplot++ REQUIRED)
endif()
add_library(nn_options INTERFACE)
target_compile_features(nn_options INTERFACE cxx_std_20)
target_include_directories(nn_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/common")
target_link_libraries(nn_options INTERFACE Eigen3::Eigen spdlog::spdlog)
target_compile_options(nn_options INTERFACE -Wall -Wextra -Wpedantic)
if(NN_SANITIZE)
  target_compile_options(nn_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nn_options INTERFACE -fsanitize=address,undefined)
endif()
foreach(hw IN ITEMS hw1 hw2 hw3 hw4)
  add_subdirectory("${hw}/src")
endforeach()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
